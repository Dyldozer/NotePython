#Requires AutoHotkey v2.0

class Kitty
{
    __New(kittenPath := "kitten.exe", listenAddress := "tcp:localhost:12345")
    {
        this.KittenPath := kittenPath
        this.ListenAddress := listenAddress
        this.WindowId := 0
    }

    ; ------------------------------------------------------------
    ; Configure which Kitty window subsequent commands target.
    ;
    ; Example:
    ;   kitty.UseWindow(17)
    ; ------------------------------------------------------------
    UseWindow(windowId)
    {
        if !IsInteger(windowId)
            throw Error("Kitty window ID must be an integer.")

        this.WindowId := windowId
        return this
    }

    ; ------------------------------------------------------------
    ; Clear the selected window.
    ; ------------------------------------------------------------
    ClearWindow()
    {
        this.WindowId := 0
        return this
    }

    ; ------------------------------------------------------------
    ; Return raw JSON from "kitten @ ls".
    ;
    ; This is useful for debugging / discovering Kitty windows.
    ; ------------------------------------------------------------
    ListJson()
    {
        return this.RunKitten(["ls"])
    }

    ; ------------------------------------------------------------
    ; Send text to the selected Kitty window.
    ;
    ; By default this method does NOT press Enter.
    ;
    ; Example:
    ;   kitty.Send("hostname")
    ;   kitty.Send("hostname`r")
    ; ------------------------------------------------------------
    Send(text, pressEnter := false)
    {
        this.RequireWindow()

        if pressEnter
            text .= "`r"

        ; -- means subsequent argument is the text.
        return this.RunKitten([
            "send-text",
            "--match",
            "id:" this.WindowId,
            text
        ])
    }

    ; ------------------------------------------------------------
    ; Send a command and press Enter.
    ;
    ; Example:
    ;   kitty.SendCommand("hostname")
    ; ------------------------------------------------------------
    SendCommand(command)
    {
        return this.Send(command, true)
    }

    ; ------------------------------------------------------------
    ; Send Ctrl+C to the remote session.
    ; ------------------------------------------------------------
    CtrlC()
    {
        this.RequireWindow()

        return this.RunKitten([
            "send-key",
            "--match",
            "id:" this.WindowId,
            "ctrl+c"
        ])
    }

    ; ------------------------------------------------------------
    ; Get text currently visible on the screen.
    ; ------------------------------------------------------------
    GetScreen()
    {
        this.RequireWindow()

        return this.RunKitten([
            "get-text",
            "--match",
            "id:" this.WindowId,
            "--extent",
            "screen"
        ])
    }

    ; ------------------------------------------------------------
    ; Get the output from the most recently executed command.
    ;
    ; This requires Kitty shell integration.
    ; ------------------------------------------------------------
    GetLastOutput()
    {
        this.RequireWindow()

        return this.RunKitten([
            "get-text",
            "--match",
            "id:" this.WindowId,
            "--extent",
            "last_cmd_output"
        ])
    }

    ; ------------------------------------------------------------
    ; Get all terminal text including scrollback.
    ; ------------------------------------------------------------
    GetAllText()
    {
        this.RequireWindow()

        return this.RunKitten([
            "get-text",
            "--match",
            "id:" this.WindowId,
            "--extent",
            "all"
        ])
    }

    ; ------------------------------------------------------------
    ; Get the currently selected text in Kitty.
    ; ------------------------------------------------------------
    GetSelection(clearSelection := false)
    {
        this.RequireWindow()

        args := [
            "get-text",
            "--match",
            "id:" this.WindowId,
            "--extent",
            "selection"
        ]

        if clearSelection
            args.Push("--clear-selection")

        return this.RunKitten(args)
    }

    ; ------------------------------------------------------------
    ; Get screen text repeatedly until a pattern appears.
    ;
    ; Useful for waiting for:
    ;
    ;   Password:
    ;   $
    ;   #
    ;   some expected output
    ;
    ; Returns the terminal screen when the pattern is found.
    ; Throws on timeout.
    ; ------------------------------------------------------------
    WaitFor(pattern, timeoutMs := 5000, pollMs := 100)
    {
        this.RequireWindow()

        start := A_TickCount

        while (A_TickCount - start < timeoutMs)
        {
            text := this.GetScreen()

            if RegExMatch(text, pattern)
                return text

            Sleep(pollMs)
        }

        throw TimeoutError(
            "Timed out waiting for pattern: " pattern
        )
    }

    ; ------------------------------------------------------------
    ; Send a command, then wait for a regex to appear.
    ;
    ; Example:
    ;
    ;   output := kitty.CommandAndWait(
    ;       "hostname",
    ;       "[$#>] "
    ;   )
    ;
    ; NOTE:
    ; This is screen/prompt based. For reliable command completion,
    ; GetLastOutput() + shell integration is preferable.
    ; ------------------------------------------------------------
    CommandAndWait(command, pattern, timeoutMs := 5000)
    {
        this.SendCommand(command)

        return this.WaitFor(pattern, timeoutMs)
    }

    ; ------------------------------------------------------------
    ; Execute an arbitrary kitten @ command.
    ;
    ; Example:
    ;
    ;   kitty.RunKitten([
    ;       "get-text",
    ;       "--match", "id:17",
    ;       "--extent", "screen"
    ;   ])
    ; ------------------------------------------------------------
    RunKitten(args)
    {
        command := this.BuildCommand(args)

        shell := ComObject("WScript.Shell")

        exec := shell.Exec(command)

        stdout := exec.StdOut.ReadAll()
        stderr := exec.StdErr.ReadAll()

        exitCode := exec.ExitCode

        if exitCode != 0
        {
            throw Error(
                "Kitty remote-control command failed.`n`n"
                "Command:`n"
                command
                "`n`nError:`n"
                stderr
            )
        }

        return stdout
    }

    ; ------------------------------------------------------------
    ; Internal: construct:
    ;
    ; kitten @ --to tcp:localhost:12345 command ...
    ; ------------------------------------------------------------
    BuildCommand(args)
    {
        command := this.Quote(this.KittenPath)
        command .= " @"
        command .= " --to " this.Quote(this.ListenAddress)

        for arg in args
            command .= " " this.Quote(arg)

        return command
    }

    ; ------------------------------------------------------------
    ; Windows command-line quoting.
    ; ------------------------------------------------------------
    Quote(value)
    {
        ; Empty argument
        if value = ""
            return '""'

        ; Quote arguments containing whitespace or shell-sensitive chars.
        if RegExMatch(value, '[\s"&|<>^]') || InStr(value, "'")
        {
            ; Escape embedded double quotes for Windows command line.
            value := StrReplace(value, '"', '\"')
            return '"' value '"'
        }

        return value
    }

    ; ------------------------------------------------------------
    ; Make sure a Kitty window has been selected.
    ; ------------------------------------------------------------
    RequireWindow()
    {
        if !this.WindowId
            throw Error(
                "No Kitty window selected.`n"
                "Call kitty.UseWindow(windowId) first."
            )
    }
}


class TimeoutError extends Error
{
    __New(message)
    {
        super.__New(message)
        this.Name := "TimeoutError"
    }
}
