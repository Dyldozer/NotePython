#Requires AutoHotkey v2.0

global TargetHwnd := 0
global Transparency := 180 ; 0-255 (lower = more transparent)

; Ctrl+Alt+T: Toggle click-through mode on the active window
^!t::
{
    global TargetHwnd, Transparency

    hwnd := WinExist("A")

    ; If we're toggling off the current target
    if (hwnd = TargetHwnd)
    {
        WinSetTransparent(255, "ahk_id " hwnd)

        exStyle := WinGetExStyle("ahk_id " hwnd)
        WinSetExStyle(exStyle & ~0x20, "ahk_id " hwnd) ; Remove WS_EX_TRANSPARENT

        TargetHwnd := 0
        ToolTip("Click-through OFF")
        SetTimer(() => ToolTip(), -1000)
        return
    }

    ; Disable previous target if one exists
    if (TargetHwnd)
    {
        try
        {
            WinSetTransparent(255, "ahk_id " TargetHwnd)
            exStyle := WinGetExStyle("ahk_id " TargetHwnd)
            WinSetExStyle(exStyle & ~0x20, "ahk_id " TargetHwnd)
        }
    }

    ; Enable on new target
    WinSetTransparent(Transparency, "ahk_id " hwnd)

    exStyle := WinGetExStyle("ahk_id " hwnd)
    WinSetExStyle(exStyle | 0x20, "ahk_id " hwnd) ; WS_EX_TRANSPARENT

    TargetHwnd := hwnd

    ToolTip("Click-through ON")
    SetTimer(() => ToolTip(), -1000)
}

; While Ctrl is held, temporarily disable click-through
~Ctrl::
{
    global TargetHwnd

    if (!TargetHwnd)
        return

    exStyle := WinGetExStyle("ahk_id " TargetHwnd)
    WinSetExStyle(exStyle & ~0x20, "ahk_id " TargetHwnd)
}

; Re-enable click-through when Ctrl is released
~Ctrl Up::
{
    global TargetHwnd

    if (!TargetHwnd)
        return

    exStyle := WinGetExStyle("ahk_id " TargetHwnd)
    WinSetExStyle(exStyle | 0x20, "ahk_id " TargetHwnd)
}
