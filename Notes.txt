#Requires AutoHotkey v2.0
#SingleInstance Force

; ============================================================
; Tutorial Maker - AHK v2 + PowerShell
; ============================================================

global Recording := false
global FrameCount := 0
global OutputDir := ""
global CapturePending := false

; ------------------------------------------------------------
; SETTINGS
; ------------------------------------------------------------

FrameDelay := 800        ; GIF frame duration in milliseconds
CaptureDelay := 150      ; Wait after click before screenshot

; ------------------------------------------------------------
; START RECORDING
; ------------------------------------------------------------

F8::
{
    global Recording, FrameCount, OutputDir

    if Recording
        return

    FrameCount := 0

    ; Create recording folder on Desktop
    desktop := A_Desktop

    timestamp := FormatTime(, "yyyy-MM-dd_HH-mm-ss")
    OutputDir := desktop "\Tutorial_" timestamp

    DirCreate(OutputDir)

    Recording := true

    ToolTip("🔴 RECORDING`n`nLeft-click to capture steps`nF9 = Finish`nEsc = Cancel")

    SetTimer(HideToolTip, -2500)
}

; ------------------------------------------------------------
; CAPTURE AFTER LEFT CLICK
; ------------------------------------------------------------

~LButton Up::
{
    global Recording, CapturePending, CaptureDelay

    if !Recording
        return

    ; Prevent several captures from being queued at once.
    if CapturePending
        return

    CapturePending := true

    SetTimer(CaptureFrame, -CaptureDelay)
}

; ------------------------------------------------------------
; TAKE SCREENSHOT
; ------------------------------------------------------------

CaptureFrame()
{
    global Recording, FrameCount, OutputDir, CapturePending

    CapturePending := false

    if !Recording
        return

    FrameCount++

    filename := Format("{:04}.png", FrameCount)
    filepath := OutputDir "\" filename

    ; PowerShell screenshot
    psCommand := "
    (
Add-Type -AssemblyName System.Windows.Forms
Add-Type -AssemblyName System.Drawing

$bounds = [System.Windows.Forms.Screen]::PrimaryScreen.Bounds

$bitmap = New-Object System.Drawing.Bitmap $bounds.Width, $bounds.Height
$graphics = [System.Drawing.Graphics]::FromImage($bitmap)

$graphics.CopyFromScreen(
    $bounds.Left,
    $bounds.Top,
    0,
    0,
    $bounds.Size
)

$bitmap.Save('%FILE%', [System.Drawing.Imaging.ImageFormat]::Png)

$graphics.Dispose()
$bitmap.Dispose()
    )"

    psCommand := StrReplace(psCommand, "%FILE%", filepath)

    RunWait(
        'powershell.exe -NoProfile -ExecutionPolicy Bypass -Command "' 
        . StrReplace(psCommand, '"', '\"')
        . '"',
        ,
        "Hide"
    )

    ToolTip("🔴 Step " FrameCount " captured")
    SetTimer(HideToolTip, -700)
}

; ------------------------------------------------------------
; STOP + CREATE GIF
; ------------------------------------------------------------

F9::
{
    global Recording, FrameCount, OutputDir, FrameDelay

    if !Recording
        return

    Recording := false

    if FrameCount = 0
    {
        ToolTip("No screenshots were captured.")
        SetTimer(HideToolTip, -2000)
        return
    }

    ToolTip("Creating GIF...`nPlease wait.")

    gifPath := OutputDir "\tutorial.gif"

    ; PowerShell script to create animated GIF
    psCommand := "
    (
Add-Type -AssemblyName System.Drawing

$folder = '%FOLDER%'
$output = '%OUTPUT%'
$delay = %DELAY%

$files = Get-ChildItem $folder -Filter '*.png' |
    Sort-Object Name

if ($files.Count -eq 0) {
    exit
}

$images = @()

foreach ($file in $files) {
    $images += [System.Drawing.Image]::FromFile($file.FullName)
}

$encoder = [System.Drawing.Imaging.ImageCodecInfo]::GetImageEncoders() |
    Where-Object { $_.MimeType -eq 'image/gif' }

$first = $images[0]

$propertyItem = [System.Runtime.Serialization.FormatterServices]::GetUninitializedObject(
    [System.Drawing.Imaging.PropertyItem]
)

$propertyItem.Id = 0x5100
$propertyItem.Type = 4
$propertyItem.Len = 4

$delayValue = [Math]::Max(1, [Math]::Round($delay / 10))

$bytes = New-Object byte[] 4
[BitConverter]::GetBytes([int]$delayValue).CopyTo($bytes, 0)
$propertyItem.Value = $bytes

$first.SetPropertyItem($propertyItem)

$encoderParams = New-Object System.Drawing.Imaging.EncoderParameters 1
$encoderParams.Param[0] = New-Object System.Drawing.Imaging.EncoderParameter(
    [System.Drawing.Imaging.Encoder]::SaveFlag,
    [int][System.Drawing.Imaging.EncoderValue]::MultiFrame
)

$first.Save($output, $encoder, $encoderParams)

$encoderParams.Param[0] = New-Object System.Drawing.Imaging.EncoderParameter(
    [System.Drawing.Imaging.Encoder]::SaveFlag,
    [int][System.Drawing.Imaging.EncoderValue]::FrameDimensionTime
)

foreach ($image in $images[1..($images.Count - 1)]) {

    $image.SetPropertyItem($propertyItem)

    $first.SaveAdd($image, $encoderParams)
}

$encoderParams.Param[0] = New-Object System.Drawing.Imaging.EncoderParameter(
    [System.Drawing.Imaging.Encoder]::SaveFlag,
    [int][System.Drawing.Imaging.EncoderValue]::Flush
)

$first.SaveAdd($encoderParams)

foreach ($image in $images) {
    $image.Dispose()
}

$first.Dispose()
    )"

    psCommand := StrReplace(psCommand, "%FOLDER%", OutputDir)
    psCommand := StrReplace(psCommand, "%OUTPUT%", gifPath)
    psCommand := StrReplace(psCommand, "%DELAY%", FrameDelay)

    RunWait(
        'powershell.exe -NoProfile -ExecutionPolicy Bypass -Command "'
        . StrReplace(psCommand, '"', '\"')
        . '"',
        ,
        "Hide"
    )

    ToolTip(
        "✅ Tutorial complete!`n`n"
        . FrameCount " frames captured.`n`n"
        . gifPath
    )

    SetTimer(HideToolTip, -4000)

    ; Open the recording folder
    Run(OutputDir)
}

; ------------------------------------------------------------
; CANCEL
; ------------------------------------------------------------

Esc::
{
    global Recording

    if !Recording
        return

    Recording := false

    ToolTip("Recording cancelled.")
    SetTimer(HideToolTip, -1500)
}

; ------------------------------------------------------------
; TOOLTIP
; ------------------------------------------------------------

HideToolTip()
{
    ToolTip()
}
