#Requires AutoHotkey v2.0
#SingleInstance Force

result := "ROT diagnostic`n"
result .= "================`n`n"

; First test the exact thing your script needs.
try {
    outlook := ComObjActive("Outlook.Application")
    result .= "ComObjActive Outlook: SUCCESS`n"
    result .= "Outlook Version: " outlook.Version "`n`n"
} catch as e {
    result .= "ComObjActive Outlook: FAILED`n"
    result .= "Error: " e.Message "`n"
    result .= "Code: 0x" Format("{:08X}", e.Number & 0xFFFFFFFF) "`n`n"
}

; Create a bind context.
bindCtx := 0
hr := DllCall("ole32\CreateBindCtx"
    , "UInt", 0
    , "Ptr*", &bindCtx)

if (hr != 0) {
    result .= "CreateBindCtx FAILED: 0x" Format("{:08X}", hr & 0xFFFFFFFF)
    MsgBox result
    ExitApp
}

; Get the Running Object Table.
rot := 0
hr := DllCall("ole32\GetRunningObjectTable"
    , "UInt", 0
    , "Ptr*", &rot)

if (hr != 0) {
    result .= "GetRunningObjectTable FAILED: 0x"
        . Format("{:08X}", hr & 0xFFFFFFFF)
    ObjRelease(bindCtx)
    MsgBox result
    ExitApp
}

; IRunningObjectTable::EnumRunning = vtable index 9
enum := 0
hr := ComCall(9, rot, "Ptr*", &enum)

if (hr != 0) {
    result .= "EnumRunning FAILED: 0x"
        . Format("{:08X}", hr & 0xFFFFFFFF)
    ObjRelease(rot)
    ObjRelease(bindCtx)
    MsgBox result
    ExitApp
}

result .= "Running Object Table entries:`n"
result .= "-----------------------------`n"

count := 0

Loop {
    moniker := 0
    fetched := 0

    ; IEnumMoniker::Next = vtable index 3
    hr := ComCall(3, enum
        , "UInt", 1
        , "Ptr*", &moniker
        , "UInt*", &fetched)

    ; S_FALSE = no more entries.
    if (hr != 0 || fetched = 0)
        break

    displayName := 0

    ; IMoniker::GetDisplayName = vtable index 20
    hr := ComCall(20, moniker
        , "Ptr", bindCtx
        , "Ptr", 0
        , "Ptr*", &displayName)

    if (hr = 0 && displayName) {
        name := StrGet(displayName, "UTF-16")
        result .= name "`n"

        ; Highlight anything Outlook-related.
        if InStr(name, "Outlook")
            result .= "    <--- OUTLOOK RELATED`n"

        ; CoTaskMemFree is required for GetDisplayName's returned string.
        DllCall("ole32\CoTaskMemFree", "Ptr", displayName)

        count++
    }

    ObjRelease(moniker)
}

result .= "`nTotal entries: " count "`n"

ObjRelease(enum)
ObjRelease(rot)
ObjRelease(bindCtx)

MsgBox result, "AHK v2 ROT Diagnostic"
